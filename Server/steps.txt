Code

const express = require('express');
const mongoose = require('mongoose');
const Chunk =require('../models/Chunk');
const {embedTexts} = require('../services/embeddings');

const router = express.Router();


router.post('/' ,async (req, res) => {

    try {
        const {question , documentId} =req.body;

    // Check: question missing hai ya empty hai?
    // trim() question ke aage-peeche ki spaces hatata hai.
        if(!question || !question.trim()){
            return res.status(400).json({
                error:"Question is required"
            });
        }
        // sawaal ko bhi vector mein badlo (taskType QUERY)
            // Question ko embedding vector mein convert karo.
    // RETRIEVAL_QUERY batata hai ki ye search query hai,
    // PDF ka document text nahi.
    // [queryVector] pehle returned vector ko nikaalta hai.
        const [queryVector] = await embedTexts([question], "RETRIEVAL_QUERY");
      
        const vectorStage = {
            index : "vector_index",
            path:"embedding",
            queryVector,
            numCandidates:100,
            limit:4,
        };

        if(documentId){
            vectorStage.filter = {
                documentId: new mongoose.Types.ObjectId(documentId),
            };
        }
        
        const results = await Chunk.aggregate([
            {$vectorSearch: vectorStage},
            {
                $project: {
                    _id: 0,
                    documentId: 1,
                    chunkIndex: 1,
                    text: 1,
                    score: {$meta: "vectorSearchScore"},
                },
            },
        ]);
        res.json({
            question, 
            results 
        });
        
    } catch (error) {
       console.error(error);
       res.status(500).json({ error: "Search failed" });
        
    }
    
})
module.exports =router;
